// R-4: PEMULIHAN AKSES JALAN
// Proses per provinsi, satu per satu
// ============================================================

// -----------------------------------------------------------
// KONFIGURASI
// -----------------------------------------------------------
var NAMA_PROVINSI = 'Sumatera Barat'; // Ganti: 'Aceh' / 'Sumatera Utara' / 'Sumatera Barat'
var ASSET_JALAN   = 'projects/latihan-gee-494406/assets/osm_roads';

var periode_baseline = ['2024-08-01', '2024-10-30'];
var periode_puncak   = ['2024-11-01', '2024-12-31'];
var periode_terkini  = ['2025-01-01', '2025-05-24'];

// -----------------------------------------------------------
// 1. ROI — satu provinsi saja
// -----------------------------------------------------------
var provinsi = ee.FeatureCollection('FAO/GAUL/2015/level1')
  .filter(ee.Filter.eq('ADM0_NAME', 'Indonesia'))
  .filter(ee.Filter.stringContains('ADM1_NAME', NAMA_PROVINSI));

print('Provinsi ditemukan:', provinsi);

var roi = provinsi.geometry();

Map.centerObject(roi, 8);

// -----------------------------------------------------------
// 2. SENTINEL-1
// -----------------------------------------------------------
var ambilS1 = function(start, end) {
  return ee.ImageCollection('COPERNICUS/S1_GRD')
    .filterBounds(roi)
    .filterDate(start, end)
    .filter(ee.Filter.eq('instrumentMode', 'IW'))
    .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VV'))
    .filter(ee.Filter.eq('orbitProperties_pass', 'DESCENDING'))
    .select('VV')
    .median()
    .clip(roi);
};

var s1_baseline = ambilS1(periode_baseline[0], periode_baseline[1]);
var s1_puncak   = ambilS1(periode_puncak[0],   periode_puncak[1]);
var s1_terkini  = ambilS1(periode_terkini[0],  periode_terkini[1]);

// -----------------------------------------------------------
// 3. MASK GENANGAN
// -----------------------------------------------------------
var threshold_abs   = -16;
var threshold_delta = -3;

var genangan_puncak = s1_puncak.lt(threshold_abs)
  .and(s1_puncak.subtract(s1_baseline).lt(threshold_delta))
  .rename('genangan_puncak');

var genangan_terkini = s1_terkini.lt(threshold_abs)
  .and(s1_terkini.subtract(s1_baseline).lt(threshold_delta))
  .rename('genangan_terkini');

// Gabung jadi satu image 2-band
var genangan_stack = genangan_puncak.addBands(genangan_terkini);


// Dari R3 -> Masukkan ke Asset
//var genangan_puncak = ee.Image(
//  'projects/latihan-gee-494406/assets/GENANGAN_PUNCAK_R3'  // sesuaikan nama asset
//).rename('genangan_puncak');

// Genangan terkini dari R-3
//var genangan_terkini = ee.Image(
//  'projects/latihan-gee-494406/assets/GENANGAN_TERKINI_R3'  // sesuaikan nama asset
//).rename('genangan_terkini');

// Gabung seperti sebelumnya
//var genangan_stack = genangan_puncak.addBands(genangan_terkini);



// -----------------------------------------------------------
// 4. DATA JALAN — filter ke ROI & jenis utama saja
// -----------------------------------------------------------
var jalan_utama = ee.FeatureCollection(ASSET_JALAN)
  .filterBounds(roi)
  .filter(ee.Filter.inList('fclass', [
    'primary', 'secondary', 'tertiary',
    'trunk', 'motorway', 'residential', 'unclassified'
  ]));

// Dari Modul 5 -> Masukkan ke Asset

// Jika Modul 5 mengekspor FeatureCollection jalan terputus sebagai asset:
//var jalan_utama = ee.FeatureCollection(
//  'projects/latihan-gee-494406/assets/JALAN_TERPUTUS_MODUL5'  // sesuaikan nama asset
//).filterBounds(roi);

// Cek filter fclass dulu di Modul 5

// -----------------------------------------------------------
// 5. BUFFER TIPIS (10m) lalu SAMPLE GENANGAN
// -----------------------------------------------------------
var jalan_dengan_status = genangan_stack.reduceRegions({
  collection: jalan_utama,
  reducer: ee.Reducer.max(), // 1 jika ada piksel genangan di sepanjang jalan
  scale: 30,
  tileScale: 4
});

// -----------------------------------------------------------
// 6. KLASIFIKASI STATUS JALAN
// -----------------------------------------------------------
var jalan_terputus = jalan_dengan_status
  .filter(ee.Filter.eq('genangan_puncak', 1));

var jalan_pulih = jalan_terputus
  .filter(ee.Filter.eq('genangan_terkini', 0));

var jalan_masih_tergenang = jalan_terputus
  .filter(ee.Filter.eq('genangan_terkini', 1));

// Jika Modul 5 sudah menglasifikasikan jalan
// jalan_terputus = langsung dari Modul 5 (tidak perlu filter lagi)

//var jalan_pulih = jalan_dengan_status
//  .filter(ee.Filter.eq('genangan_terkini', 0)); // kering sekarang = pulih

//var jalan_masih_tergenang = jalan_dengan_status
//  .filter(ee.Filter.eq('genangan_terkini', 1)); // masih basah

// -----------------------------------------------------------
// 7. HITUNG PANJANG
// -----------------------------------------------------------
var tambahPanjang = function(fc) {
  return fc.map(function(f) {
    return f.set('panjang_km', f.length().divide(1000));
  });
};

var terputus_km        = tambahPanjang(jalan_terputus);
var pulih_km           = tambahPanjang(jalan_pulih);
var masih_tergenang_km = tambahPanjang(jalan_masih_tergenang);

var total_terputus        = terputus_km.aggregate_sum('panjang_km');
var total_pulih           = pulih_km.aggregate_sum('panjang_km');
var total_masih_tergenang = masih_tergenang_km.aggregate_sum('panjang_km');
var persen_pemulihan      = total_pulih.divide(total_terputus).multiply(100);

print('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
print('HASIL R-4:', NAMA_PROVINSI);
print('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
print('Total jalan terputus saat puncak (km):', total_terputus);
print('Jalan sudah pulih / kering (km):',       total_pulih);
print('Jalan masih tergenang (km):',            total_masih_tergenang);
print('Tingkat Pemulihan R-4 (%):',             persen_pemulihan);

// -----------------------------------------------------------
// 8. VISUALISASI
// -----------------------------------------------------------
Map.addLayer(genangan_puncak.selfMask(),
  {palette: ['0000FF']}, 'Genangan Puncak');
Map.addLayer(genangan_terkini.selfMask(),
  {palette: ['00CCFF']}, 'Genangan Terkini');
Map.addLayer(jalan_pulih,
  {color: '00FF00'}, '🟢 Jalan Pulih');
Map.addLayer(
  jalan_masih_tergenang.style({
    color: 'FF6600',
    width: 1
  }),
  {},
  '🟠 Jalan Masih Tergenang'
);
Map.addLayer(jalan_terputus,
  {color: 'FF0000'}, '🔴 Semua Jalan Terputus');

// -----------------------------------------------------------
// 9. EXPORT

Export.table.toDrive({

  description: 'R4_' + NAMA_PROVINSI + '_Terputus',
  folder: 'GEE_R4_Output',
  fileFormat: 'CSV'
});

Export.table.toDrive({
  collection: pulih_km,
  description: 'R4_' + NAMA_PROVINSI + '_Pulih',
  folder: 'GEE_R4_Output',
  fileFormat: 'CSV'
});
